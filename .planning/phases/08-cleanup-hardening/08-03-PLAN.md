```go
import ("bytes"; "crypto/hmac"; "crypto/sha256"; "sort")
hmacKey := func(userID, serverID string) []byte {
    m := hmac.New(sha256.New, []byte(cfg.JWTSecret)) // server-side secret as HMAC key
    m.Write([]byte(userID + ":" + serverID))
    return m.Sum(nil)[:8]
}
sort.SliceStable(servers, func(i, j int) bool {

})
```
